function kappiState(sc, u, t) {
  let pos = { x: 300, scale: 0.84 }, keys = [[0, 'stand', 'neutral']], facing = 0, bob = 0, hold = null;
  const at = (re, sid, d = 0) => wordTime(re, sid) - sc.start + d;

    case 'hook': pos = { x: 330, scale: 0.84 }; facing = 1; bob = SIT_BOB; hold = phone; keys = [[0, 'sit', 'happy']]; break;
    case 'n22': pos = { x: 330, scale: 0.84 }; keys = [[0, 'stand', 'happy'], [0.5, 'stretch', 'joy'], [2.4, 'chest', 'proud']]; break;
    case 'n0': pos = { x: 300, scale: 0.84 }; facing = 1; bob = SIT_BOB; hold = popcorn; keys = [[0, 'sitSnack', 'happy'], [at(/^Freiheit/, 'n0', -0.1), 'sitSnack', 'joy']]; break;
    case 'n2': pos = { x: 330, scale: 0.84 }; facing = 1; bob = SIT_BOB; hold = phone; keys = [[0, 'sit', 'proud'], [at(/^geklaut/, 'n2', -0.2), 'sit', 'determined']]; break;
    case 'alarm': keys = [[0, 'scared', 'surprised'], [phStart('alarm', 1, sc), 'shrug', 'worried']]; break;
    case 'w2': keys = [[0, 'think', 'thinking'], [phStart('w2', 1, sc), 'shrug', 'happy']]; break;
    case 'jetlag': pos = { x: 230, scale: 0.6 }; keys = [[0, 'shrug', 'worried']]; break;

    case 'aha': keys = [[0, 'think', 'thinking'], [at(/^leihst/, 'aha', -0.2), 'point', 'determined']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; keys = [[0, 'point', 'happy']]; break;
  }

function kappiState(sc, u, t) {
  let pos = { x: 210, scale: 0.7 }, keys = [[0, 'stand', 'neutral']], facing = 0;
  switch (sc.id) {
    case 'hook': keys = [[0, 'cheer', 'joy'], [at(/^Summen/, 'hook', sc, -0.3), 'shrug', 'worried']]; break;
    case 'now': pos = { x: 300, scale: 0.84 }; keys = [[0, 'flex', 'proud']]; break;
    case 'sound': keys = [[0, 'point', 'talk'], [at(/^künstliche/, 'sound', sc, -0.2), 'shrug', 'surprised']]; break;
    case 'fake': keys = [[0, 'point', 'talk'], [phStart('fake', 1, sc), 'chest', 'determined']]; break;
    case 'amg': keys = [[0, 'think', 'thinking'], [at(/^mehr/, 'amg', sc, -0.3), 'cheer', 'joy']]; break;
    case 'practical': pos = { x: 300, scale: 0.84 }; keys = [[0, 'point', 'talk'], [at(/^Winter/, 'practical', sc, -0.2), 'scared', 'worried']]; break;
    case 'fair': keys = [[0, 'shrug', 'neutral']]; break;
    case 'eu': pos = { x: 300, scale: 0.84 }; keys = [[0, 'point', 'talk'], [at(/^aufweichen/, 'eu', sc, -0.3), 'cheer', 'joy']]; break;
    case 'aha': keys = [[0, 'shrug', 'thinking']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; keys = [[0, 'point', 'happy']]; break;
  }

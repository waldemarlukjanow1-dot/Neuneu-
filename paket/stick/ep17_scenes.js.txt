function kappiState(sc, u, t) {
  let pos = { x: 260, scale: 0.8 }, keys = [[0, 'stand', 'neutral']], facing = 0;
  const up = CODE.has(sc.id) ? [[0, 'flex', 'joy']] : [];
  switch (sc.id) {
    case 'hook': keys = [[0, 'flex', 'neutral'], [at(/^Noch/, 'hook', sc, -0.2), 'flex', 'determined'], [phStart('hook', 1, sc), 'think', 'thinking']]; break;
    case 'rest': keys = [...up, [1.0, 'point', 'talk'], [at(/^Pause/, 'rest', sc, -0.4), 'shrug', 'happy']]; break;
    case 'sleep': keys = [...up, [1.0, 'point', 'talk'], [at(/^weniger/, 'sleep', sc, -0.3), 'scared', 'worried']]; break;
    case 'light': keys = [...up, [1.0, 'point', 'talk'], [at(/^leichte/, 'light', sc, -0.3), 'pointUp', 'determined']]; break;
    case 'protein': keys = [...up, [1.0, 'point', 'talk'], [phStart('protein', 1, sc), 'shrug', 'happy']]; break;
    case 'mind': keys = [...up, [1.0, 'think', 'thinking'], [at(/^doppelt/, 'mind', sc, -0.3), 'flex', 'surprised']]; break;
    case 'closeup': keys = [[0, 'stand', 'determined']]; break;
    case 'outro': pos = { x: 250, scale: 0.84 }; keys = [[0, 'flex', 'proud']]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.25));
  return { x: pos.x, y: 1165, scale: pos.scale, facing, pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2], muscle: level(sc, u) / 5 };
}

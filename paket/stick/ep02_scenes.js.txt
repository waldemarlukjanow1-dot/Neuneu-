function globe(x, y, r, t) {
  g.save(); g.translate(x, y);
  g.fillStyle = '#4cc9f0'; g.strokeStyle = INK; g.lineWidth = 7; g.beginPath(); g.arc(0, 0, r, 0, 7); g.fill(); g.stroke();
  g.save(); g.beginPath(); g.arc(0, 0, r - 3, 0, 7); g.clip(); g.fillStyle = '#06d6a0';
  const off = (t * 30) % (r * 4);
  for (const [cx, cy, rx, ry] of [[-0.4, -0.3, 0.35, 0.25], [0.3, -0.1, 0.3, 0.4], [-0.1, 0.45, 0.25, 0.15], [0.9, 0.3, 0.2, 0.3]]) for (const k of [-1, 0, 1]) { g.beginPath(); g.ellipse(cx * r + off - k * r * 4 - r, cy * r, rx * r, ry * r, 0.4, 0, 7); g.fill(); }
  g.restore();
  g.strokeStyle = 'rgba(22,24,29,0.35)'; g.lineWidth = 3; g.beginPath(); g.ellipse(0, 0, r * 0.45, r, 0, 0, 7); g.stroke(); g.beginPath(); g.moveTo(-r, 0); g.lineTo(r, 0); g.stroke();

}
function chartLine(x0, y0, w, h, prog, wobble, col = '#06d6a0') {
  // rising line with realistic ups and downs; prog 0..1 draws it from left to right
  g.save(); g.lineCap = 'round'; g.lineJoin = 'round';
  const pts = []; const n = 60;
  for (let i = 0; i <= n * prog; i++) { const q = i / n; const trend = Math.pow(q, 1.6); const noise = (Math.sin(q * 23) * 0.5 + Math.sin(q * 51 + 1) * 0.3 + (q > 0.45 && q < 0.55 ? -0.9 : 0)) * wobble; pts.push([x0 + q * w, y0 - (trend + noise * 0.08) * h]); }
  if (pts.length > 1) {
    g.fillStyle = 'rgba(6,214,160,0.18)'; g.beginPath(); g.moveTo(pts[0][0], y0); for (const p of pts) g.lineTo(p[0], p[1]); g.lineTo(pts[pts.length - 1][0], y0); g.closePath(); g.fill();
    g.strokeStyle = INK; g.lineWidth = 16; g.beginPath(); pts.forEach((p, i) => i ? g.lineTo(...p) : g.moveTo(...p)); g.stroke();
    g.strokeStyle = col; g.lineWidth = 9; g.beginPath(); pts.forEach((p, i) => i ? g.lineTo(...p) : g.moveTo(...p)); g.stroke();
    const e = pts[pts.length - 1]; g.fillStyle = col; g.strokeStyle = INK; g.lineWidth = 5; g.beginPath(); g.arc(e[0], e[1], 13, 0, 7); g.fill(); g.stroke();
  }
  g.strokeStyle = INK; g.lineWidth = 6; g.beginPath(); g.moveTo(x0, y0 - h - 20); g.lineTo(x0, y0); g.lineTo(x0 + w + 20, y0); g.stroke();
  g.restore();
}

  g.beginPath(); g.moveTo(420, 600); g.lineTo(1000, 1000); g.lineTo(1000, 1080); g.lineTo(420, 1080); g.closePath(); g.fill();
  g.beginPath(); g.moveTo(420, 600); g.lineTo(1000, 1000); g.stroke();
  const p = clamp01(u); const x = lerp(470, 900, p), yS = lerp(634, 931, p); const r = lerp(22, 95, Math.pow(p, 1.4));
  const cx = x, cy = yS - r * 1.2;

  g.fillStyle = '#f6c445'; g.beginPath(); g.arc(0, 0, r, 0, 7); g.fill(); g.stroke();

}
function compareBars(u, vals, maxV, label) {
  // vals: { saved, total } -> left bar: Sparschwein (saved), right bar: investiert = saved (blue) + Zinseszins (gold)
  const base = 1120, H = 520, bw = 170, xL = 540, xR = 820;
  const grow = ease(clamp01((u - 0.15) / 1.2));
  const hS = H * vals.saved / maxV * grow, hT = H * vals.total / maxV * grow;
  // axis
  g.strokeStyle = 'rgba(255,255,255,0.55)'; g.lineWidth = 4; g.setLineDash([10, 10]); for (let k = 1; k <= 4; k++) { const yy = base - H * k / 4; g.beginPath(); g.moveTo(430, yy); g.lineTo(1010, yy); g.stroke(); } g.setLineDash([]);
  const barRect = (x, h0, h, col) => { if (h <= 1) return; rr(x - bw / 2, base - h0 - h, bw, h, h0 === 0 ? 18 : 6); g.fillStyle = col; g.fill(); g.strokeStyle = INK; g.lineWidth = 6; g.stroke(); };
  barRect(xL, 0, hS, '#ff9fbf');
  barRect(xR, 0, hS, '#4cc9f0');
  barRect(xR, hS, Math.max(0, hT - hS), YEL);
  text('🐷', xL, base + 70, 56, '#000', { weight: 400, font: '"Noto Color Emoji"' });
  text('📈', xR, base + 70, 56, '#000', { weight: 400, font: '"Noto Color Emoji"' });
  text(`${fmt(vals.saved * grow)} €`, xL, base - hS - 22, 44, '#fff', { lw: 9 });
  text(`${fmt(vals.total * grow)} €`, xR, base - hT - 22, 52, YEL, { lw: 10 });

function kappiState(sc, u, t) {
  const D = sc.end - sc.start;
  let base = { x: 230, y: 1165, scale: 0.84, facing: 1 };
  let keys;
  switch (sc.id) {
    case 'hook': keys = [[0, 'hold', 'happy', 1], [D - 1.6, 'hold', 'thinking', 1], [D - 0.9, 'point', 'determined', 1]]; break;
    case 'month': keys = [[0, 'point', 'talk', 1]]; break;
    case 'year': keys = [[0, 'chest', 'happy', 0], [phStart('year', 1, sc), 'shrug', 'joy', 0]]; break;
    case 'y5': keys = [[0, 'point', 'proud', 1]]; break;
    case 'etf': keys = [[0, 'scared', 'surprised', 0], [phStart('etf', 1, sc), 'point', 'talk', 1], [phStart('etf', 2, sc), 'pointUp', 'happy', 1], [phStart('etf', 3, sc), 'think', 'thinking', 0]]; break;
    case 'y10': keys = [[0, 'pointUp', 'happy', 1]]; break;
    case 'y20': keys = [[0, 'pointUp', 'surprised', 1]]; break;
    case 'y40': keys = [[0, 'think', 'thinking', 0], [phStart('y40', 1, sc), 'scared', 'surprised', 0], [phStart('y40', 2, sc) + 0.4, 'cheer', 'joy', 0]]; break;
    case 'outro': keys = [[0, 'hold', 'happy', 1], [phStart('outro', 1, sc), 'wave', 'joy', 0]]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.3));
  return { ...base, pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2], facing: cur[3] };
}

function drawKappiFor(sc, u, t) {
  const st = kappiState(sc, u, t);
  st.blink = blink(t); st.mouth = talkingMouth(t);
  if (sc.id === 'hook') { const D = sc.end - sc.start; const gone = sm(D - 1.3, D - 1.1, u); if (gone < 1) st.hold = (gg, aR) => doener(aR.hx + 45, aR.hy - 25, 1.15 * (1 - gone), t); }
  if (sc.id === 'outro') { const p2 = phStart('outro', 1, sc); if (u < p2) st.hold = (gg, aR) => doener(aR.hx + 45, aR.hy - 25, 1.1, t); }
  return drawKappi(g, st);
}

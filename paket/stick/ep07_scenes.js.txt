// main figure placement / pose per scene: keys = [time, pose, expression]
function kappiState(sc, u, t) {
  let pos = { x: 300, scale: 0.84 }, keys = [[0, 'stand', 'neutral']], facing = 0, hold = null;
  switch (sc.id) {
    case 'hook': hold = gamepad; keys = [[0, 'hold', 'happy'], [at(/^Start/, 'hook', sc, -0.3), 'cheer', 'joy'], [phStart('hook', 1, sc), 'think', 'thinking']]; break;
    case 'consoles': keys = [[0, 'point', 'talk'], [phStart('consoles', 1, sc), 'shrug', 'worried']]; break;
    case 'server': keys = [[0, 'think', 'thinking'], [at(/^Absturz/, 'server', sc, -0.3), 'scared', 'surprised']]; break;
    case 'power': keys = [[0, 'point', 'talk'], [at(/^Hälfte/, 'power', sc, -0.2), 'scared', 'surprised']]; break;
    case 'streets': pos = { x: 230, scale: 0.7 }; keys = [[0, 'shrug', 'neutral'], [at(/^Pizzabäcker/, 'streets', sc, -0.2), 'shrug', 'worried']]; break;
    case 'money': keys = [[0, 'point', 'happy'], [wordTime(/^GTA/, 'money', 1) - sc.start - 0.2, 'cheer', 'joy']]; break;
    case 'aha': keys = [[0, 'think', 'thinking'], [at(/^normaler/, 'aha', sc, -0.2), 'shrug', 'happy']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; hold = null; keys = [[0, 'point', 'happy']]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.3));
  return { x: pos.x, y: 1165, scale: pos.scale, facing, hold, pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2] };
}

// main figure placement / pose per scene: keys = [time, pose, expression]
function kappiState(sc, u, t) {
  let pos = { x: 300, scale: 0.84 }, keys = [[0, 'stand', 'neutral']], facing = 0, hold = null;
  switch (sc.id) {
    case 'hook': hold = gamepad; keys = [[0, 'hold', 'happy'], [at(/^irgendwann/, 'hook', sc, -0.2), 'hold', 'joy']]; break;
    case 'gtp': keys = [[0, 'scared', 'surprised'], [at(/^Geräusche/, 'gtp', sc, -0.2), 'think', 'thinking']]; break;
    case 'money': keys = [[0, 'cheer', 'joy'], [at(/^Bestätigt/, 'money', sc, -0.2), 'shrug', 'worried']]; break;
    case 'spiral': hold = gamepad; keys = [[0, 'hold', 'determined'], [at(/^Kühlschrank/, 'spiral', sc, -0.2), 'hold', 'sick']]; break;
    case 'addiction': keys = [[0, 'pointUp', 'determined']]; break;
    case 'signs': keys = [[0, 'think', 'worried']]; break;
    case 'aha': keys = [[0, 'stretch' in POSES ? 'stretch' : 'cheer', 'joy']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; keys = [[0, 'point', 'happy']]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.3));
  return { x: pos.x, y: 1165, scale: pos.scale, facing, hold, pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2] };
}
function sceneBack(sc, u, t) {
  if (sc.id === 'hook') {   // the game screen grows until it fills the stage
    const tI = at(/^irgendwann/, 'hook', sc), gs = ease(clamp01((u - tI) / 1.6));
    monitor(lerp(720, 600, gs), lerp(800, 820, gs), lerp(300, 820, gs), lerp(190, 560, gs), t, gs);

}

function drawKappiFor(sc, u, t) {
  const st = kappiState(sc, u, t);
  st.blink = blink(t); st.mouth = talkingMouth(t);
  return drawKappi(g, st);
}

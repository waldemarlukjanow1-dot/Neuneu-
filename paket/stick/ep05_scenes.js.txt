function analogClock(x, y, r, h, label) {   // h = hours as float (e.g. 2.5 = 2:30)
  g.save(); g.translate(x, y);
  g.fillStyle = '#fff'; g.strokeStyle = INK; g.lineWidth = 9; g.beginPath(); g.arc(0, 0, r, 0, 7); g.fill(); g.stroke();
  for (let i = 0; i < 12; i++) { const a = i / 12 * Math.PI * 2; g.lineWidth = i % 3 ? 4 : 8; g.beginPath(); g.moveTo(Math.sin(a) * r * 0.78, -Math.cos(a) * r * 0.78); g.lineTo(Math.sin(a) * r * 0.9, -Math.cos(a) * r * 0.9); g.stroke(); }
  const am = (h % 1) * Math.PI * 2, ah = ((h % 12) / 12) * Math.PI * 2; g.lineCap = 'round';
  g.lineWidth = r * 0.07; g.beginPath(); g.moveTo(0, 0); g.lineTo(Math.sin(am) * r * 0.72, -Math.cos(am) * r * 0.72); g.stroke();
  g.lineWidth = r * 0.1; g.beginPath(); g.moveTo(0, 0); g.lineTo(Math.sin(ah) * r * 0.48, -Math.cos(ah) * r * 0.48); g.stroke();
  g.fillStyle = INK; g.beginPath(); g.arc(0, 0, r * 0.08, 0, 7); g.fill();
  g.restore();
  if (label) { g.font = '900 34px Inter'; const w = g.measureText(label).width + 40; rr(x - w / 2, y + r + 16, w, 56, 28); g.fillStyle = INK; g.fill(); text(label, x, y + r + 56, 34, YEL); }
}

function kappiState(sc, u, t) {
  let pos = { x: 300, scale: 0.84 }, keys = [[0, 'stand', 'neutral']], facing = 0;
  switch (sc.id) {
    case 'hook': pos = { x: 260, scale: 0.84 }; keys = [[0, 'think', 'thinking'], [1.2, 'scared', 'surprised']]; break;
    case 'gift': keys = [[0, 'stand', 'happy'], [at(/^schlafen/, 'gift', sc, -0.6), 'cheer', 'joy']]; break;
    case 'body': keys = [[0, 'shrug', 'worried'], [phStart('body', 1, sc), 'point', 'talk']]; facing = 0; break;
    case 'vote': keys = [[0, 'cheer', 'joy'], [1.6, 'point', 'happy']]; break;
    case 'choice': pos = { x: 540, scale: 0.84 }; keys = [[0, 'shrug', 'thinking']]; break;
    case 'summer': keys = [[0, 'wave', 'joy']]; break;
    case 'dark': keys = [[0, 'scared', 'worried']]; break;
    case 'expert': keys = [[0, 'pointUp', 'determined'], [phStart('expert', 1, sc), 'scared', 'surprised']]; break;
    case 'eu': keys = [[0, 'point', 'talk'], [at(/^Länder/, 'eu', sc, -0.2), 'shrug', 'worried']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; keys = [[0, 'point', 'happy']]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.3));
  return { x: pos.x, y: 1165, scale: pos.scale, facing, pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2] };
}

      break;
    }
    case 'outro': {
      withScale(690, 600, pop(u, 0.3) * (1 + 0.02 * Math.sin(t * 5)), () => { card(-290, -120, 580, 240, '#fff'); text('Team ☀️ oder', 0, -30, 60, INK, { font: EMO }); text('Team 🌙? 👇', 0, 55, 60, '#ef476f', { font: EMO }); });
      g.globalAlpha = sm(0.5, 1.2, u);
      const parts = TL.sources.split(' · ');
      text(parts.slice(0, 2).join(' · '), 540, 1500, 28, 'rgba(255,255,255,0.95)', { weight: 700, lw: 6 });
      text(parts.slice(2).join(' · '), 540, 1542, 28, 'rgba(255,255,255,0.95)', { weight: 700, lw: 6 });
      g.globalAlpha = 1;
      break;
    }
  }
}

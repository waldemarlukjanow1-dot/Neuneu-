      break;
    }
    case 'outro': {
      withScale(690, 600, pop(u, 0.3) * (1 + 0.02 * Math.sin(t * 5)), () => { card(-290, -120, 580, 240, '#fff'); text('Rauchfrei-Hotline', 0, -35, 44, INK); text('0800 8 31 31 31', 0, 30, 50, '#1f4fd6'); text('kostenlos · BIÖG', 0, 80, 26, '#666', { weight: 800 }); });

      g.globalAlpha = sm(0.5, 1.2, u);
      const parts = TL.sources.split(' · ');

      g.globalAlpha = 1;
      break;
    }

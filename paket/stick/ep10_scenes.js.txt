// keys = [time, pose, expression] for Wowa (W) and Nilja (N)
function poseKeys(sc, u) {
  let W = [[0, 'stand', 'neutral']], N = [[0, 'stand', 'neutral']];
  const ph = (k) => phStart(sc.id, k, sc);
  switch (sc.id) {
    case 'hook': W = [[0, 'chest', 'happy']]; N = [[0, 'stand', 'neutral'], [ph(1) - 0.1, 'pointUp', 'determined']]; break;
    case 'gullu': W = [[0, 'point', 'talk'], [at(/^Pennerkatze/, 'gullu', sc, -0.3), 'shrug', 'joy'], [ph(2) - 0.1, 'point', 'happy']]; N = [[0, 'stand', 'neutral'], [ph(1) - 0.1, 'scared', 'surprised'], [ph(1) + 0.8, 'pointUp', 'determined'], [ph(3) - 0.1, 'chest', 'proud']]; break;
    case 'nala': W = [[0, 'point', 'talk'], [at(/^hübsch/, 'nala', sc, -0.3), 'shrug', 'happy']]; N = [[0, 'stand', 'happy'], [ph(1) - 0.1, 'cheer', 'joy']]; break;
    case 'chaos': W = [[0, 'scared', 'surprised'], [ph(1) - 0.1, 'shrug', 'thinking']]; N = [[0, 'stand', 'worried'], [ph(2) - 0.1, 'shrug', 'happy']]; break;
    case 'score': W = [[0, 'pointUp', 'determined']]; N = [[0, 'stand', 'neutral'], [ph(1) - 0.1, 'point', 'determined']]; break;
    case 'twist': W = [[0, 'sit', 'happy']]; N = [[0, 'stand', 'joy']]; break;
    case 'outro': W = [[0, 'wave', 'happy']]; N = [[0, 'wave', 'joy']]; break;
  }
  const pick = (keys) => { let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++; const cur = keys[k], prev = keys[Math.max(0, k - 1)]; const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.3)); return { pose: mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2] }; };
  return { W: pick(W), N: pick(N) };
}

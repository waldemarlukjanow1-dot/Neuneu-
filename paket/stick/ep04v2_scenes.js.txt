// ---------- camera: punch-ins and shake (zoom keeps the focus point in place) ----------
function camera(sc, u, t) {
  const D = sc.end - sc.start, shake = (amt) => [(hash(Math.floor(t * 30)) - 0.5) * amt, (hash(Math.floor(t * 30) + 7) - 0.5) * amt];
  let keys = [[0, 540, 1000, 1]], dx = 0, dy = 0, drift = 0.03 * sm(0, D, u);
  switch (sc.id) {
    case 'hook': { const tN = at(/^Noch/, 'hook', sc); keys = [[0, ...FACE_SIT, 1]]; drift = u < tN ? 0.18 * sm(0, tN, u) : 0; if (u > tN && u < tN + 0.45) [dx, dy] = shake(40); break; }
    case 'n0': keys = [[0, 540, 1000, 1], [at(/^Klar/, 'n0', sc, -0.1), 290, 846, 1.35, 0.18]]; break;
    case 'n2': keys = [[0, ...FACE_SIT, 1]]; drift = 0.16 * sm(0, D, u); break;
    case 'tease': keys = [[0, 540, 1000, 1], [at(/^fies/, 'tease', sc, -0.15), ...FACE_STAND, 1.45, 0.18]]; break;
    case 'alarm': if (u < 1.5) [dx, dy] = shake(36 * (1 - u / 1.5)); break;
    case 'w2': keys = [[0, 540, 1000, 1], [phStart('w2', 1, sc) + 0.1, ...FACE_STAND, 1.3, 0.3]]; break;
    case 'immune': { const tV = at(/^viermal/, 'immune', sc); if (u > tV && u < tV + 0.6) [dx, dy] = shake(30); break; }
    case 'aha': { const tL = at(/^leihst/, 'aha', sc); keys = [[0, 540, 1000, 1], [tL - 0.35, ...PENDANT, 1.9, 0.35], [tL + 1.35, 540, 1000, 1, 0.4]]; drift = 0; break; }
    case 'loop': drift = 0; keys = [[0, ...FACE_SIT, 1]]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const b = k === 0 ? 1 : ease(clamp01((u - cur[0]) / (cur[4] || 0.35)));
  return { fx: lerp(prev[1], cur[1], b), fy: lerp(prev[2], cur[2], b), z: lerp(prev[3], cur[3], b) + drift, dx, dy };
}

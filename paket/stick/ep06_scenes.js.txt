// main figure placement / pose per scene: keys = [time, pose, expression]
function kappiState(sc, u, t) {
  let pos = { x: 300, scale: 0.84 }, keys = [[0, 'stand', 'neutral']], facing = 0, extraPose = null;
  switch (sc.id) {
    case 'hook': keys = [[0, 'stand', 'neutral'], [at(/^2016|^zweitausend/, 'hook', sc, -0.3), 'scared', 'surprised']]; break;
    case 'wake': keys = [[0, 'cheer', 'joy'], [phStart('wake', 0, sc) + 2.4, 'point', 'happy']]; break;
    case 'apps': keys = [[0, 'shrug', 'neutral'], [phStart('apps', 1, sc), 'chest', 'joy']]; break;
    case 'games': {
      const tD = at(/^Dafür/, 'games', sc);
      if (u > tD) { facing = 1; const w = u - tD; pos = { x: lerp(220, 420, clamp01(w / 3)), scale: 0.84 }; extraPose = Object.assign({}, POSES.stand, walkPose(w * 7, 0.6)); }
      keys = [[0, 'shrug', 'worried'], [tD, 'stand', 'joy']]; break;
    }
    case 'school': keys = [[0, 'point', 'happy'], [at(/^Dab/, 'school', sc, -0.25), 'dab', 'determined']]; break;
    case 'freeze': keys = [[0, 'stand', 'surprised'], [at(/^erstarren/, 'freeze', sc, -0.25), 'cheer', 'joy']]; break;
    case 'summer': keys = [[0, 'cheer', 'joy']]; break;
    case 'aha': keys = [[0, 'scared', 'surprised'], [at(/^Dabei/, 'aha', sc, -0.2), 'think', 'thinking']]; break;
    case 'outro': pos = { x: 290, scale: 0.84 }; facing = 1; keys = [[0, 'point', 'happy']]; break;
  }
  let k = 0; while (k + 1 < keys.length && u >= keys[k + 1][0]) k++;
  const cur = keys[k], prev = keys[Math.max(0, k - 1)];
  const bl = k === 0 ? 1 : ease(clamp01((u - cur[0]) / 0.25));
  return { x: pos.x, y: 1165, scale: pos.scale, facing, pose: extraPose || mixPose(POSES[prev[1]], POSES[cur[1]], bl), expr: bl > 0.5 ? cur[2] : prev[2] };
}
